document.addEventListener('DOMContentLoaded', async () => { // --- AUTHENTICATION & USER HELPERS --- function getCurrentUser() { try { return JSON.parse(localStorage.getItem('loggedInUser')); } catch (e) { return null; } } const user = getCurrentUser(); if (!user || !user.username) { window.location.href = 'index.html'; return; } function getUserKey(keyName) { return `${user.username}_${keyName}`; } // --- DOM ELEMENTS --- const avatarPreview = document.getElementById('avatar-preview'); const avatarInput = document.getElementById('avatar-input'); const displayPlayerName = document.getElementById('displayPlayerName'); const openEditNameModal = document.getElementById('openEditNameModal'); const editNameModal = document.getElementById('editNameModal'); const closeEditNameModal = document.getElementById('closeEditNameModal'); const usernameInput = document.getElementById('username-input'); const saveProfileBtn = document.getElementById('save-profile-btn'); const saveStatus = document.getElementById('save-status'); const displayLevelBadge = document.getElementById('displayLevelBadge'); const displayXpText = document.getElementById('displayXpText'); const displayXpBarFill = document.getElementById('displayXpBarFill'); // --- 1. FETCH & SYNC DATA FROM FIRESTORE --- if (window.pixvinzDb) { const { db, doc, getDoc } = window.pixvinzDb; try { const userDocRef = doc(db, "players", user.username); const docSnap = await getDoc(userDocRef); if (docSnap.exists()) { const data = docSnap.data(); if (data.displayName) { user.displayName = data.displayName; localStorage.setItem('loggedInUser', JSON.stringify(user)); } if (data.avatar) { localStorage.setItem(getUserKey('vinpix_avatar'), data.avatar); } if (data.level !== undefined) { localStorage.setItem(getUserKey('currentLevel'), data.level); } if (data.coins !== undefined) { localStorage.setItem(getUserKey('totalCoins'), data.coins); } } } catch (error) { console.error("Error fetching profile from cloud:", error); } } // --- 2. POPULATE INITIAL PROFILE UI --- const currentName = user.displayName || user.username; if (displayPlayerName) displayPlayerName.innerText = currentName; if (usernameInput) usernameInput.value = currentName; // Load Avatar function loadAvatarUI() { const savedAvatar = localStorage.getItem(getUserKey('vinpix_avatar')) || localStorage.getItem('vinpix_avatar'); if (savedAvatar && avatarPreview) { avatarPreview.src = savedAvatar; } } loadAvatarUI(); // Load Level & XP Progress const currentLevel = parseInt(localStorage.getItem(getUserKey('currentLevel'))) || 1; if (displayLevelBadge) displayLevelBadge.innerText = `LEVEL ${currentLevel}`; // XP Calculation simulation based on level/coins const totalCoins = parseInt(localStorage.getItem(getUserKey('totalCoins'))) || 0; const currentXp = (currentLevel - 1) * 100 + (totalCoins % 100); const maxXp = currentLevel * 100; const xpPercentage = Math.min(100, Math.max(0, (currentXp / maxXp) * 100)); if (displayXpText) displayXpText.innerText = `${currentXp} / ${maxXp} XP`; if (displayXpBarFill) displayXpBarFill.style.width = `${xpPercentage}%`; // --- 3. HANDLE AVATAR FILE UPLOAD & CLOUD SYNC --- if (avatarInput) { avatarInput.addEventListener('change', (e) => { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async function(uploadEvent) { const base64String = uploadEvent.target.result; localStorage.setItem(getUserKey('vinpix_avatar'), base64String); loadAvatarUI(); // Save to Firestore Cloud immediately if (window.pixvinzDb) { const { db, doc, setDoc } = window.pixvinzDb; try { await setDoc(doc(db, "players", user.username), { username: user.username, displayName: user.displayName || user.username, level: currentLevel, coins: totalCoins, avatar: base64String, lastUpdated: new Date() }, { merge: true }); } catch (err) { console.error("Failed to sync avatar to cloud:", err); } } }; reader.readAsDataURL(file); }); } // --- 4. EDIT NAME MODAL CONTROLS --- if (openEditNameModal && editNameModal) { openEditNameModal.addEventListener('click', () => { editNameModal.classList.remove('hidden'); if (usernameInput) usernameInput.value = user.displayName || user.username; if (saveStatus) saveStatus.innerText = ''; }); } if (closeEditNameModal && editNameModal) { closeEditNameModal.addEventListener('click', () => { editNameModal.classList.add('hidden'); }); } if (editNameModal) { editNameModal.addEventListener('click', (e) => { if (e.target === editNameModal) { editNameModal.classList.add('hidden'); } }); } // --- 5. SAVE PROFILE NAME & CLOUD SYNC --- if (saveProfileBtn) { saveProfileBtn.addEventListener('click', async () => { const newName = usernameInput.value.trim(); if (!newName) { if (saveStatus) { saveStatus.style.color = '#ff5252'; saveStatus.innerText = "Name cannot be empty!"; } return; } // Update LocalStorage User user.displayName = newName; localStorage.setItem('loggedInUser', JSON.stringify(user)); // Update registeredUsers registry let registeredUsers = JSON.parse(localStorage.getItem('registeredUsers')) || {}; if (registeredUsers[user.username]) { registeredUsers[user.username].displayName = newName; localStorage.setItem('registeredUsers', JSON.stringify(registeredUsers)); } if (displayPlayerName) displayPlayerName.innerText = newName; if (saveStatus) { saveStatus.style.color = '#4caf50'; saveStatus.innerText = "Saving to cloud..."; } // Sync updates to Firestore Cloud if (window.pixvinzDb) { const { db, doc, setDoc } = window.pixvinzDb; try { const savedAvatar = localStorage.getItem(getUserKey('vinpix_avatar')) || ''; await setDoc(doc(db, "players", user.username), { username: user.username, displayName: newName, level: currentLevel, coins: totalCoins, avatar: savedAvatar, lastUpdated: new Date() }, { merge: true }); if (saveStatus) { saveStatus.innerText = "Successfully saved!"; setTimeout(() => { editNameModal.classList.add('hidden'); }, 1000); } } catch (error) { console.error("Error saving profile to cloud:", error); if (saveStatus) { saveStatus.style.color = '#ff5252'; saveStatus.innerText = "Saved locally, cloud sync failed."; } } } else { if (saveStatus) { saveStatus.innerText = "Successfully saved!"; setTimeout(() => { editNameModal.classList.add('hidden'); }, 1000); } } }); } }); // Global back function referenced in profile.html function goHome() { window.location.href = 'index.html'; }
